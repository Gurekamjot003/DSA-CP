QUICK(ARR, N, BEG, END, LOC)
...

1.  [initialize] 
    LEFTPTR = BEG
    RIGHTPTR = END
    LOC = BEG

2.  [Scan ARR from right to left until RIGHTPTR>LOC]
    (a) repeat while ARR[RIGHTPTR] >= ARR[LOC] and RIGHTPTR > LOC:
            RIGHTPTR = RIGHTPTR-1 
        [end of loop]
    (b) if LOC = RIGHTPTR, then RETURN
    (c) if ARR[RIGHTPTR] < ARR[LOC], then:
        (i)     [interchange the values]
                set TEMP = ARR[RIGHTPTR]
                ARR[RIGHTPTR] = ARR[LOC]
                ARR[LOC] = TEMP
        (ii)    [set new location]
                set LOC = RIGHTPTR
        (iii)   go to step 3

3.  [scan ARR from left to right until LEFTPTR < LOC]
    (a) repeat while ARR[LEFTPTR] <= ARR[LOC] and LEFTPTR < LOC:
            LEFTPTR = LEFTPTR - 1
        [end of loop]
    (b) if LEFTPTR = LOC, then RETURN.
    (c) if ARR[LEFTPTR] > ARR[LOC], then:
        (i)     [interchange the values]
                set TEMP = ARR[LEFTPTR]
                ARR[LEFTPTR] = ARR[LOC]
                ARR[LOC] = TEMP
        (ii)    [reset LOC]
                set LOC = LEFTPTR
        (iii)   go to step 2
