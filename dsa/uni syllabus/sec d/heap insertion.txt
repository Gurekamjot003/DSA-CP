INSHEAP(TREE, N, ITEM)

1.  [initialize]
    set N = N + 1
    PTR = N

2.  repeat steps 3 to 5 while PTR > 1

3.  set PAR = PTR/2

4.  if ITEM <= TREE[PAR], then:
        set TREE[PTR] = ITEM, and return
    [end of if structure]

5.  set TREE[PTR] = TREE[PAR]
    PTR = PAR
[end of step 2 loop]

6.  set TREE[1] = ITEM

7.  Exit.
